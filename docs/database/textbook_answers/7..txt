//tuple-based / attribute-based checks are not checked during deletions, and do not hold for subqueries
    - e.g You can insert duplicate address despite a check like CHECK(address NOT IN (...))
    because SQL does not require DBMS to evaluate subqueries as table-wide invariants.
    Transaction order, isolation, deffered evaluation can all allow duplicates to slip in
//trigger-based checks are checked when specified by programmer
//assertion-based checks are checked during updates, deletions, or insertions

(c) ASSERTION more_than_9_sinks_less_than_9
    CHECK (
        NOT EXISTS(
            SELECT s1.name, s2.name
            FROM (
                SELECT s.name AS name, o.battle AS battle
                FROM Classes c 
                JOIN Ships s ON c.class = s.class
                JOIN Outcomes o ON o.ship = s.name
                WHERE c.numGuns > 9 
            ) s1
            JOIN (
                SELECT s.name AS name, o.battle AS battle
                FROM Classes c
                JOIN Ships s ON c.class = s.class
                JOIN Outcomes o ON o.ship = s.name AND result = 'sunk'
                WHERE c.numGuns < 9
            ) s2 
            ON s1.battle = s2.battle
        )
    )
